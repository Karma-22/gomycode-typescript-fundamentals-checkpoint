Interface Vehicle & Implémentation de la classe Car avec typage strict.

📜 Code TypeScript Source (src/vehicle.ts)

export interface Vehicle {
  make: string;
  model: string;
  year: number;
  start(): void;
}

export class Car implements Vehicle {
  constructor(
    public make: string,
    public model: string,
    public year: number
  ) {}

  public start(): void {
    console.log("Car engine started");
  }
}
